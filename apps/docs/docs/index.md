    root.setAttribute("data-active", id || pinned || "");
    revealNote(id);
  }
  // Brings a note into view inside the notes column — never the page, which
  // `scrollIntoView` would also move, taking the pointer off the marker.
  function revealNote(id) {
    var list = document.querySelector(".mx-home-notes");
    var note = id && document.getElementById("note-" + id);
    if (!list || !note || list.scrollHeight <= list.clientHeight) return;
    if (note.offsetTop < list.scrollTop) list.scrollTop = note.offsetTop - 8;
    else if (note.offsetTop + note.offsetHeight > list.scrollTop + list.clientHeight) {
      list.scrollTop = note.offsetTop + note.offsetHeight - list.clientHeight + 8;
    }
  }
  document.addEventListener("mouseover", function (event) { show(markerId(event.target)); });
  document.addEventListener("mouseout", function (event) { if (!markerId(event.relatedTarget)) show(""); });
  document.addEventListener("focusin", function (event) { show(markerId(event.target)); });
  document.addEventListener("focusout", function () { show(""); });
  document.addEventListener("keydown", function (event) {
    var id = markerId(event.target);
    if (!id) return;
    if (event.key === "Enter" || event.key === " " || event.key === "Spacebar") {
      event.preventDefault();
      pinned = id;
      show(id);
    } else if (event.key === "Escape") {
      pinned = "";
      show("");
    }
  });
  document.addEventListener("click", function (event) {
    var id = markerId(event.target);
    if (id) { pinned = id; show(id); return; }
    if (!event.target.closest || !event.target.closest(".mx-home-theme")) return;
    var next = document.documentElement.getAttribute("data-theme") === "dark" ? "light" : "dark";
    document.documentElement.setAttribute("data-theme", next);
    document.body.setAttribute("data-theme", next);
    try { localStorage.setItem("docmd-theme", next); } catch (error) { /* private mode */ }
  });
})();
